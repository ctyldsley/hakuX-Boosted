# hakuX-Boosted: tests that fail on the ARM64 back end as it is on
# claude/integration-1113, with every translator setting at its default.
# They are skipped in CI so that a red run always means a new difference.
# One test name a line, as in ../skip.txt. "name unless <flag>" skips the
# test only in runs that are not given that flag; "name if <flag>" only in
# runs that are.

# stwcx. to an address other than the one lwarx reserved must fail without
# storing. With a64_native_reserved_ops (the default) the store-conditional
# does not compare addresses and stores. Titles pair the two on one address.
stwcx_address_mismatch

# mcrf between fields other than from cr0 is wrong as upstream translates
# it; ppc_mcrf_copies_field, on by default since claude/jit-difftest (the
# differential test's reference agrees with it on every case), translates
# it exactly. These fail only in a run that turns the setting off.
mcrf_fields_cr7_to_cr3 if --ppc_mcrf_copies_field=false
mcrf_fields_gt if --ppc_mcrf_copies_field=false
mcrf_fields_zero_source if --ppc_mcrf_copies_field=false

# Upstream's own mcrf tests that copy from a field other than cr0 expect what
# upstream's translation gives (a compare of the source field against zero),
# not a copy of the field: mcrf_3 expects "greater" in cr0 after copying a
# field that holds "equal". They fail with the exact translation, which is
# now the default, and are skipped in every run.
mcrf_3
mcrf_3_constant
mcrf_5
mcrf_5_constant

# Hand-written flag conventions the ABI does not have (a callee returning
# its answer in cr0, a callee taking a flag as an argument): level 3 of
# eliminate_dead_context_stores gets them wrong without its guard, which is
# on by default, so these are skipped only where a line turns it off. The
# last one makes every later function keep its flags, and runs only in the
# line that names the guard.
cr_returned_by_callee_with_branch if --dead_context_stores_abi_guard=false
cr_returned_through_jump if --dead_context_stores_abi_guard=false
cr_argument_to_callee if --dead_context_stores_abi_guard=false
cr_argument_through_stub if --dead_context_stores_abi_guard=false
cr_returned_by_callee_indirect unless --dead_context_stores_abi_guard=true
